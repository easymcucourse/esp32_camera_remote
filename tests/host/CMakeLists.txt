cmake_minimum_required(VERSION 3.16)
project(camera_parsers C)
enable_testing()
add_executable(test_camera_parsers test_camera_parsers.c
    ../../components/ptpip/ptp_dataset.c ../../components/sony_camera/sony_props.c)
target_include_directories(test_camera_parsers PRIVATE
    ../../components/ptpip/include ../../components/sony_camera/include)
set_property(TARGET test_camera_parsers PROPERTY C_STANDARD 11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_camera_parsers PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME camera_parsers COMMAND test_camera_parsers)

add_executable(test_focus_input test_focus_input.c ../../main/focus_input.c)
target_include_directories(test_focus_input PRIVATE ../../main)

add_executable(test_focus_caps test_focus_caps.c ../../components/sony_camera/sony_props.c)
target_include_directories(test_focus_caps PRIVATE ../../components/ptpip/include ../../components/sony_camera/include)
add_test(NAME focus_caps COMMAND test_focus_caps)
add_executable(test_sony_write test_sony_write.c ../../components/sony_camera/sony_ext.c)
target_include_directories(test_sony_write PRIVATE stubs ../../components/ptpip/include ../../components/sony_camera/include)
add_test(NAME sony_write COMMAND test_sony_write)
foreach(target test_focus_input test_focus_caps test_sony_write)
    set_property(TARGET ${target} PROPERTY C_STANDARD 11)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
